###########################################
# 1789 Diplomacy — opening wars and scripted treaties
#
# These wars are intentionally bounded. Historical treaty timers close them
# automatically, while on_capitulation_immediate applies the same settlement
# early if a participant collapses.
###########################################

add_namespace = napoleonic_diplomacy

# ============================================================
# THEATRE WAR — Denmark-Norway leaves the Swedish war, 9 July 1789
# ============================================================
country_event = {
	id = napoleonic_diplomacy.1
	title = napoleonic_diplomacy.1.t
	desc = napoleonic_diplomacy.1.d
	picture = GFX_report_event_generic_assembly
	fire_only_once = yes

	trigger = {
		tag = DEN
		date > 1789.7.8
		has_global_flag = nap_theatre_war_active
		has_war_with = SWE
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = napoleonic_diplomacy.1.a
		napoleonic_end_theatre_war = yes
		news_event = { id = napoleonic_diplomacy.101 hours = 6 }
	}
}

news_event = {
	id = napoleonic_diplomacy.101

	picture = GFX_news_event_generic_assembly
	major = yes

}
