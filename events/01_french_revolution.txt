# ============================================================
# 1. STORMING OF THE BASTILLE — 1789.7.14
# ============================================================
country_event = {
	id = french_revolution.1
	title = french_revolution.1.t
	desc = french_revolution.1.d
	picture = GFX_report_event_generic_riot

	fire_only_once = yes

	trigger = {
		tag = FRA
		date > 1789.7.13
		has_country_flag = bourbon_monarchy
		NOT = { has_country_flag = bastille_stormed }
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = french_revolution.1.a         # "Vive la liberté!"
		set_country_flag = bastille_stormed
		add_political_power = -50
		add_stability = -0.15
		add_war_support = 0.10

		# Garde Nationale forms (Lafayette commands)
		load_oob = "FRA_garde_nationale"

		news_event = { id = napoleonic_timeline.1 days = 1 }
	}

	option = {
		name = french_revolution.1.b         # "Crush the rebels!" — alt-history hardline
		set_country_flag = bourbon_crackdown
		set_country_flag = french_royalist_path
		clr_country_flag = french_constitutional_path
		clr_country_flag = french_revolutionary_path
		clr_country_flag = french_bonapartist_path
		add_political_power = -100
		add_stability = -0.30
		add_war_support = -0.10
		# Locks the player into a counter-revolutionary path
	}
}

# 6. REPUBLIC PROCLAIMED — 1792.9.22
# ============================================================
# Triggered either by date or by the "Vive la République" focus (which can fire it earlier).
country_event = {
	id = french_revolution.8
	title = french_revolution.8.t
	desc = french_revolution.8.d
	picture = GFX_report_event_generic_assembly

	fire_only_once = yes

	trigger = {
		tag = FRA
		OR = {
			date > 1792.9.21
			has_country_flag = vive_la_republique_focus
		}
		has_country_flag = first_coalition_war
		has_country_flag = french_revolutionary_path
		NOT = { has_country_flag = french_constitutional_path }
		NOT = { has_country_flag = french_royalist_path }
		NOT = { has_country_flag = republic_proclaimed }
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = french_revolution.8.a         # "Vive la République!"
		set_country_flag = republic_proclaimed
		set_politics = {
			ruling_party = communism
			elections_allowed = no
		}
		add_popularity = { ideology = communism popularity = 0.40 }
		add_popularity = { ideology = neutrality popularity = -0.40 }
		remove_ideas = bourbon_crisis
		add_ideas = revolutionary_republic
		add_war_support = 0.20
		add_stability = -0.10                # internal turmoil
		clr_country_flag = bourbon_monarchy
		set_country_flag = french_republic

		news_event = { id = napoleonic_timeline.8 hours = 6 }

		# Triggers the king's trial sequence
		country_event = { id = french_revolution.9 days = 90 }
	}
}

# 7. EXECUTION OF LOUIS XVI — 1793.1.21
# ============================================================
# Fires automatically 90 days after the Republic is proclaimed (or by focus).
# This is the event the user specifically requested: "Vive la révolution" focus → king's death.
country_event = {
	id = french_revolution.9
	title = french_revolution.9.t
	desc = french_revolution.9.d
	picture = GFX_report_event_generic_riot

	fire_only_once = yes

	trigger = {
		tag = FRA
		has_country_flag = french_republic
		has_country_flag = french_revolutionary_path
		NOT = { has_country_flag = french_constitutional_path }
		NOT = { has_country_flag = french_royalist_path }
		OR = {
			has_country_flag = vive_la_revolution_focus
			date > 1793.1.20
		}
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = french_revolution.9.a         # "Le roi est mort. Vive la République!"
		set_country_flag = louis_xvi_executed
		retire_country_leader = yes          # Removes Louis XVI from leader pool
		add_war_support = 0.10
		add_political_power = 50
		add_stability = -0.15
		add_popularity = { ideology = communism popularity = 0.15 }
		add_popularity = { ideology = neutrality popularity = -0.10 }

		# Diplomatic consequences — Europe in horror
		every_country = {
			limit = {
				OR = { tag = ENG tag = HAB tag = PRU tag = RUS tag = SPR tag = NAP tag = POR }
				NOT = { has_war_with = FRA }
			}
			country_event = { id = french_revolution.91 days = 2 }
		}

		news_event = { id = napoleonic_timeline.9 hours = 6 }
	}
}
