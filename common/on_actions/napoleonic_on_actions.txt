# Napoleonic Era - startup hooks for HOI4 1.19.x.
# State ownership is applied at runtime to avoid duplicate-state CTDs caused by
# version-specific vanilla history/states filenames.
#
# ROOT is the capitulated country and FROM is the winner in
# on_capitulation_immediate.

on_actions = {
	on_startup = {
		effect = {
			FRA = {
				napoleonic_setup_1789_map = yes
				napoleonic_setup_1789_diplomacy = yes
				news_event = { id = napoleonic_timeline.0 days = 1 }
			}
		}
	}

	# The opening wars are intentionally limited conflicts. Intercept
	# capitulations before HOI4 can turn them into normal total-war peace
	# conferences and apply the bounded treaty settlement instead.
	on_capitulation_immediate = {
		effect = {
			# Denmark-Norway / Sweden: Theatre War.
			if = {
				limit = {
					has_global_flag = nap_theatre_war_active
					OR = {
						AND = { ROOT = { tag = DEN } FROM = { tag = SWE } }
						AND = { ROOT = { tag = SWE } FROM = { tag = DEN } }
					}
				}
				napoleonic_end_theatre_war = yes
				DEN = { news_event = { id = napoleonic_diplomacy.101 days = 0 } }
			}

			# If Sweden collapses in either northern war, end both Scandinavian
			# conflicts before a standard peace conference can annex Sweden.
			if = {
				limit = { ROOT = { tag = SWE } }
				if = {
					limit = { has_global_flag = nap_russo_swedish_war_active }
					napoleonic_end_russo_swedish_war = yes
					SWE = { news_event = { id = napoleonic_diplomacy.102 days = 0 } }
				}
				if = {
					limit = { has_global_flag = nap_theatre_war_active }
					napoleonic_end_theatre_war = yes
					DEN = { news_event = { id = napoleonic_diplomacy.101 days = 0 } }
				}
			}

			# A Russian collapse ends both of Russia's opening wars. Sweden gets
			# status quo; the Ottomans avoid the historical Jassy cession.
			if = {
				limit = { ROOT = { tag = RUS } }
				if = {
					limit = { has_global_flag = nap_russo_swedish_war_active }
					napoleonic_end_russo_swedish_war = yes
					SWE = { news_event = { id = napoleonic_diplomacy.102 days = 0 } }
				}
				if = {
					limit = { has_global_flag = nap_russo_turkish_war_active }
					napoleonic_end_russo_turkish_ottoman_success = yes
					TUR = { news_event = { id = napoleonic_diplomacy.105 days = 0 } }
				}
			}

			# An Ottoman collapse ends both southern wars. Russia receives only
			# the bounded Jassy settlement; Austria receives no whole-state gain.
			if = {
				limit = { ROOT = { tag = TUR } }
				if = {
					limit = { has_global_flag = nap_austro_turkish_war_active }
					napoleonic_end_austro_turkish_war = yes
					HAB = { news_event = { id = napoleonic_diplomacy.103 days = 0 } }
				}
				if = {
					limit = { has_global_flag = nap_russo_turkish_war_active }
					napoleonic_end_russo_turkish_jassy = yes
					RUS = { news_event = { id = napoleonic_diplomacy.104 days = 0 } }
				}
			}

			# If Austria alone is forced out, settle only its Ottoman war.
			if = {
				limit = {
					ROOT = { tag = HAB }
					has_global_flag = nap_austro_turkish_war_active
				}
				napoleonic_end_austro_turkish_war = yes
				HAB = { news_event = { id = napoleonic_diplomacy.103 days = 0 } }
			}
		}
	}
}
