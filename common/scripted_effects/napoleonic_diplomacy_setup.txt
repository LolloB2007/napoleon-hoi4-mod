napoleonic_end_theatre_war = {
 if = { limit = { has_global_flag = nap_theatre_war_active DEN = { has_war_with = SWE } NOT = { has_global_flag = nap_opening_peace_busy } }
  set_global_flag = nap_opening_peace_busy
  clr_global_flag = nap_theatre_war_active
  set_global_flag = nap_theatre_war_ended
  DEN = { clr_country_flag = theatre_war_active }
  SWE = { clr_country_flag = theatre_war_active }
  if = { limit = { DEN = { has_war_with = SWE } } DEN = { white_peace = SWE } }
  
  DEN = { add_stability = 0.03 news_event = { id = napoleonic_diplomacy.101 hours = 6 } }
  clr_global_flag = nap_opening_peace_busy
 }
}
