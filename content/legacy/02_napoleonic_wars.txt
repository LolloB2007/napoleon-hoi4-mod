# ============================================================
# EMPIRE PROCLAIMED — 1804.5.18
# ============================================================
country_event = {
	id = napoleonic_wars.3
	title = napoleonic_wars.3.t
	desc = napoleonic_wars.3.d
	picture = GFX_report_event_generic_assembly

	fire_only_once = yes

	trigger = {
		tag = FRA
		has_country_flag = consulate_established
		has_country_flag = french_bonapartist_path
		NOT = { has_country_flag = french_constitutional_path }
		NOT = { has_country_flag = french_royalist_path }
		OR = {
			date > 1804.5.17
			has_country_flag = proclaim_empire_focus
		}
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = napoleonic_wars.3.a           # "L'Empire est proclamé!"
		ai_chance = { factor = 95 }
		set_country_flag = empire_of_the_french
		clr_country_flag = consulate_established
		remove_ideas = revolutionary_republic
		add_ideas = empire_of_the_french
		add_popularity = { ideology = fascism popularity = 0.20 }
		add_war_support = 0.15
		add_stability = 0.10
		add_political_power = 100

		country_event = { id = napoleonic_wars.4 days = 200 }    # Coronation Dec 2
	}

	option = {
		name = napoleonic_wars.3.b           # Refuse the crown — Republic continues
		ai_chance = { factor = 5 }
		add_stability = -0.05
		add_political_power = -50
	}
}

	}

	}

	mean_time_to_happen = { days = 1 }

		name = napoleonic_wars.16.a          # March!
