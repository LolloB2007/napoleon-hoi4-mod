add_namespace = napoleonic_collapse

# ============================================================
# RETREAT FROM MOSCOW — 1812.10.18
# ============================================================
country_event = {
	id = napoleonic_collapse.1
	title = napoleonic_collapse.1.t
	desc = napoleonic_collapse.1.d
	picture = GFX_report_event_generic_riot

	fire_only_once = yes

	trigger = {
		tag = FRA
		has_country_flag = russian_campaign_active
		date > 1812.10.17
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = napoleonic_collapse.1.a       # The Grande Armée begins to die
		# Apply massive attrition and the Russian Winter idea
		add_ideas = russian_winter
		add_war_support = -0.30
		add_political_power = -100
		add_command_power = -50

		# Auto-fires the army-loss event chain (not implemented in detail here)
		set_country_flag = grande_armee_destroyed

		news_event = { id = napoleonic_collapse.11 hours = 6 }
	}
}

# ============================================================
# WAR OF LIBERATION — 1813.3.16
# ============================================================
# Prussia rises. Sixth Coalition forms.
country_event = {
	id = napoleonic_collapse.2
	title = napoleonic_collapse.2.t
	desc = napoleonic_collapse.2.d
	picture = GFX_report_event_generic_army_formed

	fire_only_once = yes

	trigger = {
		tag = PRU
		FRA = { has_country_flag = grande_armee_destroyed }
		date > 1813.3.15
	}

	mean_time_to_happen = { days = 1 }

	option = {
		name = napoleonic_collapse.2.a       # The Eagle and the Iron Cross
		ai_chance = { factor = 95 }
		add_ideas = landwehr_call_to_arms
		add_war_support = 0.30
		declare_war_on = { target = FRA type = annex_everything }

		# Rally allies
		every_country = {
			limit = {
				OR = { tag = RUS tag = HAB tag = SWE tag = ENG }
				NOT = { has_war_with = FRA }
			}
			add_opinion_modifier = { target = PRU modifier = huge_increase }
		}
	}

	option = {
		name = napoleonic_collapse.2.b       # Wait
		ai_chance = { factor = 5 }
	}
}

# ============================================================
# LEIPZIG — Battle of the Nations — 1813.10.16-19
# ============================================================
news_event = {
	id = napoleonic_collapse.3
	title = napoleonic_collapse.3.t
	desc = napoleonic_collapse.3.d
	picture = GFX_news_event_generic_army_formed

	major = yes

	trigger = {
		date > 1813.10.15
		FRA = { has_war_with = PRU }
		NOT = { has_global_flag = leipzig_fought }
	}

	mean_time_to_happen = { days = 1 }
	fire_only_once = yes

	option = {
		name = napoleonic_collapse.3.a       # The Empire begins to collapse
		set_global_flag = leipzig_fought
		FRA = {
			add_war_support = -0.30
			add_political_power = -150
			add_command_power = -100
		}
		# Lose the German clients — un-puppet at top scope so PREV/THIS scoping is clean.
		every_country = {
			limit = {
				OR = { tag = BAV tag = WUR tag = BAD tag = HES tag = SAX }
				is_subject_of = FRA
			}
			# THIS = the German state being iterated. End its puppet status with FRA.
			FRA = { end_puppet = PREV }
		}
		PRU = { add_war_support = 0.20 }
		HAB = { add_war_support = 0.20 }
		RUS = { add_war_support = 0.20 }
	}
}

# ============================================================
# NAPOLEON ABDICATES — Fontainebleau — 1814.4.11
# ============================================================
country_event = {
	id = napoleonic_collapse.5
	title = napoleonic_collapse.5.t
	desc = napoleonic_collapse.5.d
	picture = GFX_report_event_generic_riot

	fire_only_once = yes

	trigger = {
		tag = FRA
		date > 1814.4.10
		has_country_flag = empire_of_the_french
		OR = {
			has_war_with = PRU
			has_war_with = HAB
			has_war_with = RUS
			has_war_with = ENG
		}
	}

	mean_time_to_happen = { days = 7 }

	option = {
		name = napoleonic_collapse.5.a       # Abdicate
		ai_chance = { factor = 80 }
		retire_country_leader = yes          # Napoleon to Elba
		set_country_flag = napoleon_on_elba
		clr_country_flag = empire_of_the_french
		remove_ideas = empire_of_the_french

		# White peace with the coalition
		every_country = {
			limit = { has_war_with = FRA }
			FRA = { white_peace = PREV }
		}

		# Restoration follows in 22 days
		country_event = { id = napoleonic_collapse.6 days = 22 }
	}

	option = {
		name = napoleonic_collapse.5.b       # Fight on to the end
		ai_chance = { factor = 20 }
		add_war_support = 0.05
		add_stability = -0.10
	}
}

# THE HUNDRED DAYS — Napoleon returns from Elba — 1815.3.1
# ============================================================
country_event = {
	id = napoleonic_collapse.8
	title = napoleonic_collapse.8.t
	desc = napoleonic_collapse.8.d
	picture = GFX_report_event_generic_army_formed

	fire_only_once = yes

	trigger = {
		tag = FRA
		has_country_flag = bourbon_restoration
		has_country_flag = napoleon_on_elba
		OR = {
			date > 1815.2.28
			has_country_flag = return_from_elba_focus
		}
	}

	mean_time_to_happen = { days = 14 }

	option = {
		name = napoleonic_collapse.8.a       # "L'Aigle volera de clocher en clocher!"
		ai_chance = { factor = 80 }
		set_country_flag = hundred_days
		clr_country_flag = bourbon_restoration
		clr_country_flag = napoleon_on_elba
		remove_ideas = bourbon_restoration_idea
		set_politics = {
			ruling_party = fascism
			elections_allowed = no
		}
		add_popularity = { ideology = fascism popularity = 0.50 }
		add_popularity = { ideology = neutrality popularity = -0.30 }
		add_ideas = liberal_empire

		# Coalition immediately reforms
		every_country = {
			limit = {
				OR = { tag = ENG tag = PRU tag = HAB tag = RUS }
			}
			declare_war_on = { target = FRA type = annex_everything }
		}

		news_event = { id = napoleonic_collapse.81 hours = 6 }
		country_event = { id = napoleonic_collapse.9 days = 109 }    # Waterloo June 18
	}

	option = {
		name = napoleonic_collapse.8.b       # Stay on Elba
		ai_chance = { factor = 20 }
	}
}
